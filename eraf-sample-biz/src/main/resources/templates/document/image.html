// 이미지 리사이즈 (비율 유지)
ImageUtils.resize(sourcePath, targetPath, 400, 300);

// 강제 리사이즈
ImageUtils.resizeForce(sourcePath, targetPath, 400, 300);

// 썸네일 생성
ImageUtils.thumbnail(sourcePath, targetPath, 150);

// 텍스트 워터마크 (투명도 0.5)
ImageUtils.watermark(sourcePath, targetPath, "ERAF Sample", 0.5f);

// 이미지 회전
ImageUtils.rotate(sourcePath, targetPath, 90);

// 이미지 정보 조회
ImageUtils.ImageInfo info = ImageUtils.getInfo(path);
System.out.println(info.width() + "x" + info.height());

// 포맷 변환
