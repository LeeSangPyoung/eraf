#8 Barcode 생성

바코드가 여기에 표시됩니다

#9 QR Code 생성

QR 코드가 여기에 표시됩니다

Code Example 
// Barcode 생성
BarcodeGenerator.generateCode128("ERAF-123", Path.of("barcode.png"), 300, 100);

// 바이트 배열로 생성
byte[] barcodeBytes = BarcodeGenerator.generateCode128ToBytes("ERAF-123", 300, 100);

// QR Code 생성
QRCodeGenerator.generate("https://example.com", Path.of("qr.png"), 300);

// QR 바이트 배열
byte[] qrBytes = QRCodeGenerator.generateToBytes("https://example.com", 300);

// vCard QR
QRCodeGenerator.generateVCard("John Doe", "010-1234-5678", "john@example.com", Path.of("vcard.png"), 300);

// WiFi QR
QRCodeGenerator.generateWifi("MySSID", "password", "WPA", Path.of("wifi.png"), 300);
