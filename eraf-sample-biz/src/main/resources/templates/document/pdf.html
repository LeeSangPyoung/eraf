Code Example 
// HTML to PDF 변환
String html = """
    <html>
    <body>
        <h1>Hello PDF</h1>
        <p>This is a PDF document.</p>
    </body>
    </html>
    """;

// 파일로 저장
PdfGenerator.fromHtml(html, Path.of("output.pdf"));

// 바이트 배열로 반환
byte[] pdfBytes = PdfGenerator.fromHtmlToBytes(html, null);

// 간단한 텍스트 PDF
PdfGenerator.createTextPdf("Simple text content", Path.of("text.pdf"));
